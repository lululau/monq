# Monq Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Pry workbench that loads user-authored Mongoid models from `~/.monq.d` and queries MongoDB the way Arql queries SQL.

**Architecture:** `Cli` parses options, `App` merges YAML and owns environments, `Environment` owns one Mongoid client plus one namespace. Display (`t`/`v`/`vd`) and Pry commands sit on top and never open connections themselves.

**Tech Stack:** Ruby >= 3.2, Mongoid ~> 9, mongo driver, Pry, net-ssh-gateway, terminal-table, rainbow, caxlsx, roo, RSpec.

**Spec:** `docs/superpowers/specs/2026-10-24-monq-design.md`

# frozen_string_literal: true

require "yaml"
require "ostruct"

module Monq
  module Config
    RESERVED = %w[default arql].freeze

    module_function

    def load_file(path)
      raw = YAML.safe_load(File.read(path), aliases: true)
      (raw || {}).transform_keys(&:to_s)
    rescue ArgumentError
      raw = YAML.safe_load(File.read(path))
      (raw || {}).transform_keys(&:to_s)
    end

    def select_environments(yaml, names, options)
      yaml = stringify(yaml)
      if names.nil? || names.empty?
        return { "default" => cli_env(options) } if sufficient_cli?(options)

        raise Error, "No -e given and CLI has no connection. Pass -e or --uri, or create ~/.monq.d/init.yml"
      end

      names.each do |name|
        raise Error, "#{name} is a reserved environment name" if RESERVED.include?(name)
        raise Error, "Environment #{name} not in config. Known: #{known(yaml).join(', ')}" unless yaml.key?(name)
      end

      selected = names.to_h { |n| [n, deep_sym(yaml[n])] }
      apply_cli!(selected, options)
      assign_namespaces!(selected)
      assert_unique_namespaces!(selected)
      selected
    end

    def known(yaml)
      yaml.keys - RESERVED
    end

    def sufficient_cli?(options)
      options.uri.to_s != "" || (options.host.to_s != "" && options.database.to_s != "")
    end

    def cli_env(options)
      {
        uri: options.uri,
        host: options.host,
        port: options.port,
        database: options.database,
        username: options.username,
        password: options.password,
        auth_source: options.auth_source
      }.compact
    end

    def apply_cli!(selected, options)
      overrides = connection_overrides(options)
      if selected.size > 1 && overrides.any?
        raise Error, "Connection CLI flags are not allowed with multiple -e environments"
      end
      return if selected.size != 1

      env = selected.values.first
      overrides.each { |k, v| env[k] = v }
      merge_ssh!(env, options)
    end

    def connection_overrides(options)
      {
        uri: options.uri,
        host: options.host,
        port: options.port,
        database: options.database,
        username: options.username,
        password: options.password,
        auth_source: options.auth_source
      }.compact.reject { |_, v| v.respond_to?(:empty?) && v.empty? }
    end

    def merge_ssh!(env, options)
      return if options.ssh.nil? || options.ssh.empty?

      env[:ssh] = (env[:ssh] || {}).merge(options.ssh.transform_keys(&:to_sym))
    end

    def assign_namespaces!(selected)
      selected.each do |name, env|
        env[:namespace] ||= camelize(name)
      end
    end

    def assert_unique_namespaces!(selected)
      groups = selected.group_by { |_, env| env[:namespace] }
      dup = groups.find { |_, pairs| pairs.size > 1 }
      return unless dup

      raise Error, "Environments share namespace #{dup.first}: #{dup.last.map(&:first).join(', ')}"
    end

    def camelize(name)
      name.split(/[^A-Za-z0-9]+/).reject(&:empty?).map { |p| p[0].upcase + p[1..] }.join
    end

    def stringify(hash)
      hash.to_h.transform_keys(&:to_s).transform_values { |v| v.is_a?(Hash) ? stringify(v) : v }
    end

    def deep_sym(obj)
      case obj
      when Hash then obj.to_h { |k, v| [k.to_sym, deep_sym(v)] }
      else obj
      end
    end

    def prefix_for(env_name)
      env_name.split(/[._-]/, 2).first
    end

    def client_name_for(env_name)
      env_name.gsub(/[^A-Za-z0-9]/, "_").to_sym
    end
  end
end

```ruby
def client_options(env)
  env = deep_sym(env)
  ssh = env[:ssh]
  if ssh && env[:uri].to_s.start_with?("mongodb+srv:")
    raise Error, "mongodb+srv with ssh is not supported; use a direct host"
  end

  built = env[:uri].to_s != "" ? from_uri(env) : from_fields(env)
  return built unless ssh && ssh[:local_port]

  {
    hosts: ["127.0.0.1:#{ssh[:local_port]}"],
    database: built[:database],
    options: built[:options].merge(direct_connection: true).except(:replica_set, :connect)
  }
end

def from_uri(env)
  require "mongo"
  parsed = Mongo::URI.get(env[:uri])
  database = env[:database].to_s
  database = parsed.database if database.empty?
  raise Error, "database is required (URI path is empty)" if database.nil? || database.empty?

  options = parsed.uri_options.dup
  options.delete(:replica_set) if env[:ssh] && env[:ssh][:local_port]
  {
    hosts: parsed.servers,
    database: database,
    options: options
  }
end

def from_fields(env)
  raise Error, "host and database are required" if env[:host].to_s.empty? || env[:database].to_s.empty?

  port = env[:port] || 27017
  options = {}
  options[:user] = env[:username] if env[:username]
  options[:password] = env[:password] if env[:password]
  options[:auth_source] = env[:auth_source] if env[:auth_source]
  options[:replica_set] = env[:replica_set] if env[:replica_set] && !(env[:ssh] && env[:ssh][:local_port])
  options[:ssl] = env[:ssl] unless env[:ssl].nil?
  { hosts: ["#{env[:host]}:#{port}"], database: env[:database], options: options }
end

def mask_uri(uri)
  uri.to_s.sub(%r{://([^:/@]+):[^@]+@}, '://\1:***@')
end

def ssh_forward_target(env)
  env = deep_sym(env)
  if env[:uri].to_s != ""
    require "mongo"
    host, port = Mongo::URI.get(env[:uri]).servers.first.split(":")
    { host: host, port: (port || 27017).to_i }
  else
    { host: env[:host], port: (env[:port] || 27017).to_i }
  end
end
```

If `Mongo::URI.get` does not exist in the installed driver, use `Mongo::URI.new(env[:uri])` and read `#servers` / `#database` / `#options` — adjust the three method names to whatever `bundle exec ruby -rmongo -e 'p Mongo::URI.instance_methods(false)'` prints, keeping the spec expectations above.

# frozen_string_literal: true

require "mongoid"

module Monq
  class Environment
    attr_reader :name, :config, :client_name, :namespace, :client, :ssh_proxy

    def initialize(name, config, first:, model_dir:)
      @name = name
      @config = config
      @first = first
      @model_dir = model_dir
      @client_name = Config.client_name_for(name)
      @namespace = config[:namespace]
      @models = []
      @sandbox_session = nil
    end

    def connect!
      opts = Config.client_options(config)
      @client = Mongo::Client.new(opts[:hosts], **opts[:options].merge(database: opts[:database]))
      Mongoid::Clients.set(client_name, @client)
      Mongoid::Clients.set(:default, @client) if @first
      hello!
    end

    def hello!
      @client.database.command(hello: 1).documents.first
    end

    def ping?
      hello!
      true
    rescue StandardError
      false
    end

    def reconnect!
      raise Error, "sandbox is active; run sandbox-quit first" if sandbox?

      @client&.close
      connect!
    end

    def namespace_module
      @namespace_module ||= Object.const_set(namespace, Module.new)
    end

    def sandbox?
      !@sandbox_session.nil?
    end
  end
end

```ruby
def load_models!
  install_namespace_api!
  path = File.join(@model_dir, "#{Config.prefix_for(name)}.rb")
  unless File.file?(path)
    warn "model file not found: #{path}"
    return
  end
  namespace_module.module_eval(File.read(path), path)
  bind_clients!
  install_abbreviations!
end

def models
  namespace_module.constants.filter_map do |const_name|
    klass = namespace_module.const_get(const_name)
    klass if klass.is_a?(Class) && klass < Mongoid::Document && !klass.embedded?
  end
end

def redefine!
  raise Error, "sandbox is active; run sandbox-quit first" if sandbox?

  tracked_classes.each do |klass|
    Mongoid.models.delete(klass) if Mongoid.models.respond_to?(:delete)
  end
  tracked_names.each { |n| namespace_module.send(:remove_const, n) if namespace_module.const_defined?(n, false) }
  @tracked = []
  load_models!
end

def tracked_names
  @tracked ||= []
end

def tracked_classes
  tracked_names.filter_map do |n|
    namespace_module.const_get(n) if namespace_module.const_defined?(n, false)
  end.select { |k| k.is_a?(Class) }
end

def bind_clients!
  namespace_module.constants.each do |const_name|
    klass = namespace_module.const_get(const_name)
    next unless klass.is_a?(Class) && klass < Mongoid::Document
    next if klass.embedded?

    current = klass.storage_options[:client]
    next if current && current.to_sym != :default

    klass.store_in klass.storage_options.merge(client: client_name)
    tracked_names << const_name
  end
end

def install_abbreviations!
  models.each do |klass|
    abbr = klass.name.split("::").last.gsub(/[^A-Z]/, "")
    abbr = "M" if abbr.empty?
    candidate = abbr
    n = 2
    while namespace_module.const_defined?(candidate, false)
      candidate = "#{abbr}#{n}"
      n += 1
    end
    namespace_module.const_set(candidate, klass)
    tracked_names << candidate.to_sym
  end
end

def install_namespace_api!
  env = self
  mod = namespace_module
  mod.define_singleton_method(:client) { env.client }
  mod.define_singleton_method(:models) { env.models }
  mod.define_singleton_method(:collections) { env.client.database.collection_names }
  mod.define_singleton_method(:config) { env.config }
  mod.define_singleton_method(:q) { |doc| env.client.database.command(doc) }
end
```

`lib/monq/app.rb`:

```ruby
module Monq
  class App
    attr_reader :environments, :options

    class << self
      attr_accessor :instance
    end

    def initialize(options)
      @options = options
      self.class.instance = self
      @environments = build_environments
    end

    def run!
      @environments.each(&:connect!)
      load_initializer!
      @environments.each(&:load_models!)
      if @options.code && !@options.code.empty?
        eval(@options.code, TOPLEVEL_BINDING, "-e")
      elsif @options.args&.first
        load(@options.args.first)
      elsif !$stdin.isatty
        eval($stdin.read, TOPLEVEL_BINDING, "stdin")
      else
        require "monq/repl"
        Repl.new.start
      end
    end
  end
end
```

`build_environments` resolves the config file (explicit `-c` missing → raise; no file and no CLI connection → raise), calls `Config.select_environments`, and maps each to `Environment.new(..., first: index.zero?, model_dir: File.expand_path("~/.monq.d"))`.

`load_initializer!`: if `-i` set and missing, raise; if default path missing, return; else `load` once.
